```gdscript
extends Node3D
# L1: pooled sprite gib chunks. Perf contract: 48-node hard cap, oldest recycled,
# zero steady-state allocations. Wall collision = clamp_to_ring displacement.

const POOL := 48
const GRAVITY := 14.0
const DAMP := 0.45
const SPIN_MAX := 6.0

var _path = null
var _nodes: Array = []          # Sprite3D, index-aligned with _g
var _g: Array = []              # per-slot Dictionary state (pre-allocated)
var _order: Array = []          # spawn order (slot indices) for oldest-recycle

func setup(path) -> void:
    _path = path
    if _nodes.is_empty():
        var frames: Array = SpriteGen.gib_frames()
        for i in POOL:
            var s := Sprite3D.new()
            s.texture = frames[i % frames.size()]
            s.billboard = BaseMaterial3D.BILLBOARD_ENABLED
            s.shaded = false
            s.pixel_size = 0.09
            s.visible = false
            add_child(s)
            _nodes.append(s)
            _g.append({"on": false, "vel": Vector3.ZERO, "spin": 0.0,
                    "life": 0.0, "t": 0.0, "ring": 0})

func clear_all() -> void:
    for i in POOL:
        _g[i].on = false
        _nodes[i].visible = false
    _order.clear()

func active_count() -> int:
    var n := 0
    for st in _g:
        if st.on: n += 1
    return n

func burst(pos: Vector3, base_vel: Vector3, ring: int, count: int, tint: Color) -> void:
    for k in count:
        var slot := _take_slot()
        var st: Dictionary = _g[slot]
        var dir := Vector3(randf() - 0.5, randf() - 0.3, randf() - 0.5).normalized()
        st.on = true
        st.vel = base_vel * 0.4 + dir * (7.0 + randf() * 9.0)
        st.spin = (randf() * 2.0 - 1.0) * SPIN_MAX
        st.life = 1.6 + randf() * 0.8
        st.t = 0.0
        st.ring = ring
        var n: Sprite3D = _nodes[slot]
        n.global_position = pos
        n.modulate = tint
        n.scale = Vector3.ONE
        n.visible = true

func _take_slot() -> int:
    for i in POOL:
        if not _g[i].on:
            _order.append(i)
            return i
    var oldest: int = _order.pop_front()   # cap hit: recycle oldest
    _order.append(oldest)
    return oldest

func _process(delta: float) -> void:
    if get_tree().paused: return
    _sim(delta)

func _sim(dt: float) -> void:
    for i in POOL:
        var st: Dictionary = _g[i]
        if not st.on: continue
        st.t += dt
        if st.t >= st.life:
            st.on = false
            _nodes[i].visible = false
            _order.erase(i)
            continue
        st.vel.y -= GRAVITY * dt
        var n: Sprite3D = _nodes[i]
        var pos: Vector3 = n.global_position + st.vel * dt
        if _path != null:
            st.ring = _path.nearest_ring(pos, st.ring)
            var clamped: Vector3 = _path.clamp_to_ring(pos, st.ring, 0.5)
            if clamped.distance_squared_to(pos) > 0.0001:
                st.vel = st.vel.bounce((clamped - pos).normalized()) * DAMP
                pos = clamped
        n.global_position = pos
        n.rotation.z += st.spin * dt
        var frac: float = st.t / st.life
        if frac > 0.7:
            n.scale = Vector3.ONE * clamp((1.0 - frac) / 0.3, 0.05, 1.0)
```

var _intensity := 0.0
var _calm_t := 0.0
var _ramp_floor := 0.0     # set_music_intensity() writes this — the gauntlet tier ramp seam

func set_music_intensity(level: float) -> void:
    _ramp_floor = clampf(level, 0.0, 2.0)

func _update_intensity(dt: float, ppos: Vector3) -> void:
    var enemies: Node = get_node_or_null("/root/Game/EnemyManager")   # match actual tree; cache it
    var shots: Node = get_node_or_null("/root/Game/ShotManager")
    var target := 0.0
    if enemies != null:
        target += minf(enemies.near_count(ppos, 120.0) * 0.35, 1.4)
    if shots != null:
        target += minf(shots.eshot_cache.size() * 0.15, 0.6)
    target += (GameState.combo_mult() - 1) * 0.3
    target = maxf(target, _ramp_floor)
    if GameState.arena_locked: target = maxf(target, 1.0)
    if GameState.boss_active: target = 2.0
    target = clampf(target, 0.0, 2.0)
    if target >= _intensity:
        _intensity = target
        _calm_t = 0.0
    else:
        _calm_t += dt
        if _calm_t > 4.0:
            _intensity = maxf(target, _intensity - 0.5 * dt)

func _mix_weights(i: float) -> Vector3:   # (calm, combat, frenzy)
